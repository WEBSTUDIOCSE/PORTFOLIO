Fix: Stateless Rate Limiting (lib/rate-limit.ts) Upstash Redis / Vercel KV 
import { Ratelimit } from "@upstash/ratelimit";
import { Redis } from "@upstash/redis";

// Create a new ratelimiter, that allows 5 requests per 15 minutes
const ratelimit = new Ratelimit({
  redis: Redis.fromEnv(),
  limiter: Ratelimit.slidingWindow(5, "15 m"),
  analytics: true,
});

export async function rateLimit(ip: string): Promise<{ ok: boolean; retryAfter?: number }> {
  const { success, reset } = await ratelimit.limit(ip);
  if (!success) {
    const retryAfter = Math.ceil((reset - Date.now()) / 1000);
    return { ok: false, retryAfter };
  }
  return { ok: true };
}
Fix: XSS Hazard (lib/theme.tsx) Next.js Script Component 
import Script from 'next/script';

// Instead of dangerouslySetInnerHTML, use next/script beforeInteractive
export function ThemeScript() {
  return (
    <Script id="theme-flash-prevention" strategy="beforeInteractive">
      {`
        let dark;
        try { dark = localStorage.getItem("theme") === "dark" || (!localStorage.getItem("theme") && window.matchMedia("(prefers-color-scheme: dark)").matches); } catch(e) {}
        if (dark) document.documentElement.classList.add("dark");
      `}
    </Script>
  );
}
