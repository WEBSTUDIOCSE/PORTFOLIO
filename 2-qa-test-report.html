Fix: A11y (components/scroll-to-top.tsx) Add aria-label 
export default function ScrollToTop({
  className,
  children,
  "aria-label": ariaLabel = "Scroll to top",
}: {
  className?: string;
  children: React.ReactNode;
  "aria-label"?: string;
}) {
  return (
    <button
      type="button"
      aria-label={ariaLabel}
      onClick={(e) => {
        e.preventDefault();
        window.scrollTo({ top: 0, behavior: "smooth" });
      }}
      className={className}
    >
      {children}
    </button>
  );
}
